-- ============================================================================
-- 3. СИСТЕМА РОЛЕЙ ТА ДОЗВОЛІВ (ROLES, PERMISSIONS, USER_ROLES, ROLE_PERMISSIONS)   - RBAC, ABAC, ReBAC ----
-- ============================================================================
CREATE TABLE ROLES (
    ID          NUMBER GENERATED BY DEFAULT AS IDENTITY,
    CODE        VARCHAR2(50) NOT NULL,
    NAME        VARCHAR2(100) NOT NULL,
    DESCRIPTION VARCHAR2(255),

    CONSTRAINT PK_ROLES PRIMARY KEY (ID),
    CONSTRAINT UK_ROLES_CODE UNIQUE (CODE)
);

COMMENT ON TABLE ROLES IS 'Довідник доступних системних ролей (класичний RBAC).';

COMMENT ON COLUMN ROLES.CODE IS 'Унікальний алфавітно-цифровий ідентифікатор ролі для перевірки в бізнес-логіці (напр., ADMIN, AUDITOR).';
COMMENT ON COLUMN ROLES.NAME IS 'Зрозуміла назва ролі для відображення в інтерфейсі адміністрування.';


--
CREATE TABLE PERMISSIONS (
    ID          NUMBER GENERATED BY DEFAULT AS IDENTITY,
    CODE        VARCHAR2(100) NOT NULL,
    NAME        VARCHAR2(100) NOT NULL,
    DESCRIPTION VARCHAR2(255),

    CONSTRAINT PK_PERMISSIONS PRIMARY KEY (ID),
    CONSTRAINT UK_PERMISSIONS_CODE UNIQUE (CODE)
);

COMMENT ON TABLE PERMISSIONS IS 'Реєстр атомарних прав на виконання дій або доступ до ресурсів.';

COMMENT ON COLUMN PERMISSIONS.CODE IS 'Унікальний код права, зазвичай структурований через двокрапку (напр., USER:BAN, REPORT:EXPORT).';


--
CREATE TABLE USER_ROLES (
    USER_ID     NUMBER NOT NULL,
    ROLE_ID     NUMBER NOT NULL,

    CONSTRAINT PK_USER_ROLES PRIMARY KEY (USER_ID, ROLE_ID),
    CONSTRAINT FK_UR_USER FOREIGN KEY (USER_ID) REFERENCES USERS(ID) ON DELETE CASCADE,
    CONSTRAINT FK_UR_ROLE FOREIGN KEY (ROLE_ID) REFERENCES ROLES(ID) ON DELETE CASCADE
);


COMMENT ON TABLE USER_ROLES IS 'Таблиця зв''язку між користувачами та ролями. Реалізує зв''язок Багато-до-Багатьох.';


--
CREATE TABLE ROLE_PERMISSIONS (
    ROLE_ID       NUMBER NOT NULL,
    PERMISSION_ID NUMBER NOT NULL,
    CONDITION_RULE VARCHAR2(4000) DEFAULT NULL,

    CONSTRAINT PK_ROLE_PERMISSIONS PRIMARY KEY (ROLE_ID, PERMISSION_ID),
    CONSTRAINT FK_RP_ROLE FOREIGN KEY (ROLE_ID) REFERENCES ROLES(ID) ON DELETE CASCADE,
    CONSTRAINT FK_RP_PERM FOREIGN KEY (PERMISSION_ID) REFERENCES PERMISSIONS(ID) ON DELETE CASCADE,
    CONSTRAINT CHK_RP_JSON CHECK (CONDITION_RULE IS NULL OR CONDITION_RULE IS JSON)
);


COMMENT ON TABLE ROLE_PERMISSIONS IS 'Зв''язок між ролями та атомарними дозволами. Містить JSON-правила для розширення моделі до ABAC.';

COMMENT ON COLUMN ROLE_PERMISSIONS.CONDITION_RULE IS 'Динамічний контекст виконання (ABAC): JSON-опис обмежень на право (напр., робочі години, білі маски IP).';
