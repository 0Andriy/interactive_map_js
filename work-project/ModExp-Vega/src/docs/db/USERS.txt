-- ============================================================================
-- 1. СТВОРЕННЯ ТАБЛИЦІ КОРИСТУВАЧІВ (USERS)
-- ============================================================================
CREATE TABLE USERS (
    ID                     NUMBER GENERATED BY DEFAULT AS IDENTITY,
    GUID                   VARCHAR2(36) NOT NULL,
    LOGIN                  VARCHAR2(255) NOT NULL,
    EMAIL                  VARCHAR2(255) NOT NULL,
    PASSWORD_HASH          VARCHAR2(255) NOT NULL,
    LOCALE                 VARCHAR2(5) DEFAULT 'uk-UA' NOT NULL,

    -- Інфраструктура двофакторної автентифікації (2FA)
    IS_2FA_ENABLED         CHAR(1) DEFAULT 'N' NOT NULL,
    TOTP_SECRET            VARCHAR2(128) DEFAULT NULL,
    BACKUP_CODES_HASH      VARCHAR2(4000) DEFAULT NULL,

    -- Динамічні ліміти сесій для користувача
    MAX_ACTIVE_SESSIONS    NUMBER DEFAULT 5,                      -- Число, або NULL (без обмежень)

    -- Статуси життєвого циклу акаунта
    IS_ACTIVE              NUMBER(1) DEFAULT 1 NOT NULL,          -- 1 = Активний, 0 = Заблокований
    IS_EMAIL_VERIFIED      NUMBER(1) DEFAULT 0 NOT NULL,          -- 1 = Підтверджено, 0 = Ні

    -- Інфраструктура захисту від Brute-Force атак
    FAILED_LOGIN_ATTEMPTS  NUMBER DEFAULT 0 NOT NULL,             -- Лічильник помилок пароля
    LOCKED_UNTIL           TIMESTAMP DEFAULT NULL,                -- Час заморозки акаунта

    -- Часові мітки для аудиту та контролю безпеки
    LAST_LOGIN_AT          TIMESTAMP DEFAULT NULL,                -- Останній успішний вхід
    PASSWORD_CHANGED_AT    TIMESTAMP DEFAULT CURRENT_TIMESTAMP NOT NULL, -- Для застарівання паролів
    CREATED_AT             TIMESTAMP DEFAULT CURRENT_TIMESTAMP NOT NULL,
    UPDATED_AT             TIMESTAMP DEFAULT CURRENT_TIMESTAMP NOT NULL,

    -- Системні обмеження таблиці (Constraints)
    CONSTRAINT PK_USERS PRIMARY KEY (ID),
    CONSTRAINT UK_USERS_GUID UNIQUE (GUID),
    CONSTRAINT UK_USERS_LOGIN UNIQUE (LOGIN),
    CONSTRAINT UK_USERS_EMAIL UNIQUE (EMAIL),
    ONSTRAINT CHK_USERS_2FA CHECK (IS_2FA_ENABLED IN ('Y', 'N')),
    CONSTRAINT CHK_USERS_ACTIVE CHECK (IS_ACTIVE IN (0, 1)),
    CONSTRAINT CHK_USERS_EMAIL_VER CHECK (IS_EMAIL_VERIFIED IN (0, 1)),
    CONSTRAINT CHK_USERS_MAX_SESS CHECK (MAX_ACTIVE_SESSIONS IS NULL OR MAX_ACTIVE_SESSIONS > 0)
    CONSTRAINT CHK_USERS_BC_JSON CHECK (BACKUP_CODES_HASH IS NULL OR BACKUP_CODES_HASH IS JSON)
);

-- ============================================================================
-- 2. СТВОРЕННЯ ІНДЕКСІВ (Для миттєвої автентифікації на мільйонах рядків)
-- ============================================================================
-- Складений індекс для логіну та перевірки статусу (використовується при кожному вході)
CREATE INDEX IDX_USERS_LOGIN_AUTH ON USERS(LOGIN, IS_ACTIVE);

-- Індекс для входу за допомогою Email або відновлення доступу
CREATE INDEX IDX_USERS_EMAIL_AUTH ON USERS(EMAIL);

--
CREATE INDEX IDX_USERS_LOCKED ON USERS(LOCKED_UNTIL);

-- ============================================================================
-- 3. ТРИГЕР ДЛЯ АВТОМАТИЧНОГО ОНОВЛЕННЯ СТОВПЧИКА UPDATED_AT
-- ============================================================================
CREATE OR REPLACE TRIGGER TRG_USERS_UPDATED_AT
BEFORE UPDATE ON USERS
FOR EACH ROW
BEGIN
    :NEW.UPDATED_AT := CURRENT_TIMESTAMP;
END;
/
