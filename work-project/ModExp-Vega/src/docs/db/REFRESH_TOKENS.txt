Очистити лише кошик поточного користувача - PURGE RECYCLEBIN;
Видалити конкретну таблицю з кошика - PURGE TABLE REFRESH_TOKENS;


-- ============================================================================
-- 0. БЕЗПЕЧНЕ ВИДАЛЕННЯ СТАРОЇ ТАБЛИЦІ ТА ПОВ'ЯЗАНИХ ОБ'ЄКТІВ
-- ============================================================================
BEGIN
    -- 1. Видаляємо таблицю (автоматично видаляє її власну identity-послідовність)
    -- PURGE очищує кошик (Recycle Bin), щоб старий сиквенс не висів у системі
    EXECUTE IMMEDIATE 'DROP TABLE REFRESH_TOKENS CASCADE CONSTRAINTS PURGE';
EXCEPTION
    WHEN OTHERS THEN
        -- Якщо таблиці не існувало (код помилки ORA-00942), просто ігноруємо помилку
        IF SQLCODE != -942 THEN
            RAISE;
        END IF;
END;
/

-- ============================================================================
-- 1. СТВОРЕННЯ ТАБЛИЦІ СЕСІЙ (REFRESH TOKENS) - (USER_SESSIONS)
-- ============================================================================
CREATE TABLE REFRESH_TOKENS (
    ID                 NUMBER GENERATED BY DEFAULT AS IDENTITY,
    SESSION_ID         VARCHAR2(36) NOT NULL,
    USER_ID            NUMBER NOT NULL,
    USER_LOGIN         VARCHAR2(255) NOT NULL,
    APP_ID             VARCHAR2(50) NOT NULL, -- DEFAULT NULL
    TOKEN_HASH         VARCHAR2(64) NOT NULL,
    JTI                VARCHAR2(36) NOT NULL,
    PARENT_JTI         VARCHAR2(36) DEFAULT NULL,

    -- Контекст безпеки та метадані пристрою
    USER_AGENT         VARCHAR2(500),
    IP_ADDRESS         VARCHAR2(45),
    DEVICE_FINGERPRINT VARCHAR2(64) DEFAULT NULL,

    -- Метрики для аудиту та запобігання аномаліям
    ROTATION_COUNT     NUMBER DEFAULT 0 NOT NULL,

    -- Часові мітки життєвого циклу
    EXPIRES_AT         TIMESTAMP NOT NULL,
    CREATED_AT         TIMESTAMP DEFAULT CURRENT_TIMESTAMP NOT NULL,
    UPDATED_AT         TIMESTAMP DEFAULT CURRENT_TIMESTAMP NOT NULL,
    REVOKED_AT         TIMESTAMP DEFAULT NULL,

    -- Причина анулювання
    REVOKE_REASON      VARCHAR2(50) DEFAULT NULL,

    -- Визначення первинного ключа та унікальних обмежень
    CONSTRAINT PK_REFRESH_TOKENS PRIMARY KEY (ID),
    CONSTRAINT UK_RT_JTI UNIQUE (JTI),
    CONSTRAINT UK_RT_TOKEN_HASH UNIQUE (TOKEN_HASH)
);

-- ============================================================================
-- 2. СТВОРЕННЯ ІНДЕКСІВ (Критично для продуктивності Oracle без фізичних FK)
-- ============================================================================
CREATE INDEX IDX_RT_SESSION_ID ON REFRESH_TOKENS(SESSION_ID);
CREATE INDEX IDX_RT_USER_ID ON REFRESH_TOKENS(USER_ID);
CREATE INDEX IDX_RT_USER_LOGIN ON REFRESH_TOKENS(USER_LOGIN);
CREATE INDEX IDX_RT_USER_APP ON REFRESH_TOKENS(USER_ID, APP_ID);
-- CREATE INDEX IDX_RT_HASH ON REFRESH_TOKENS(TOKEN_HASH);
-- CREATE INDEX IDX_RT_JTI ON REFRESH_TOKENS(JTI);
CREATE INDEX IDX_RT_PARENT_JTI ON REFRESH_TOKENS(PARENT_JTI);
CREATE INDEX IDX_RT_FINGERPRINT ON REFRESH_TOKENS(DEVICE_FINGERPRINT);

-- ============================================================================
-- 3. ТРИГЕР ДЛЯ АВТОМАТИЧНОГО ОНОВЛЕННЯ СТОВПЧИКА UPDATED_AT
-- ============================================================================
CREATE OR REPLACE TRIGGER TRG_RT_UPDATED_AT
BEFORE UPDATE ON REFRESH_TOKENS
FOR EACH ROW
BEGIN
    :NEW.UPDATED_AT := CURRENT_TIMESTAMP;
END;
/

-- ============================================================================
-- 4. ДОКУМЕНТУВАННЯ ТАБЛИЦІ ТА КОЛОНОК (Oracle Metadata Comments)
-- ============================================================================
COMMENT ON TABLE REFRESH_TOKENS IS 'Таблиця керування сесіями користувачів (Refresh-токени) з підтримкою ротації та аудіо-контексту';

COMMENT ON COLUMN REFRESH_TOKENS.ID IS 'Унікальний інкрементний ідентифікатор запису сесії (Первинний ключ)';
COMMENT ON COLUMN REFRESH_TOKENS.SESSION_ID IS 'Ідентифікатор логічної сесії (UUID). Залишається незмінним під час ротації токенів від Login до Logout. Дозволяє керувати всією сесією';
COMMENT ON COLUMN REFRESH_TOKENS.USER_ID IS 'Числовий ідентифікатор користувача (Логічний зв''язок з первинним ключем таблиці користувачів)';
COMMENT ON COLUMN REFRESH_TOKENS.USER_LOGIN IS 'Текстовий логін користувача (Денормалізація для швидкого аудиту та бану без JOIN)';
COMMENT ON COLUMN REFRESH_TOKENS.APP_ID IS 'Ідентифікатор клієнтського додатка (напр., web-portal, ios-app, android-app) для роздільного керування сесіями';
COMMENT ON COLUMN REFRESH_TOKENS.JTI IS 'JWT ID - унікальний UUID сесії, який запікається всередину Access JWT (payload.jti)';
COMMENT ON COLUMN REFRESH_TOKENS.TOKEN_HASH IS 'Незворотний SHA-256 хеш від чистого Refresh Token UUID, що видається клієнту (Захист при витоку БД)';
COMMENT ON COLUMN REFRESH_TOKENS.PARENT_JTI IS 'JTI попереднього токена. Використовується для відстеження ланцюжка ротації (Token Rotation) та виявлення злому';
COMMENT ON COLUMN REFRESH_TOKENS.USER_AGENT IS 'Рядок User-Agent браузера чи мобільного додатку користувача для аналізу пристрою';
COMMENT ON COLUMN REFRESH_TOKENS.IP_ADDRESS IS 'IP-адреса клієнта на момент ініціалізації/останнього оновлення сесії (IPv4/IPv6)';
COMMENT ON COLUMN REFRESH_TOKENS.DEVICE_FINGERPRINT IS 'SHA-256 хеш унікального відбитка пристрою (Canvas, Hardware, Plugins) для захисту від Session Hijacking';
COMMENT ON COLUMN REFRESH_TOKENS.ROTATION_COUNT IS 'Лічильник того, скільки разів цей ланцюжок сесії оновлювався (Виявлення аномальних спам-запитів)';
COMMENT ON COLUMN REFRESH_TOKENS.EXPIRES_AT IS 'Дата та час, коли термін дії поточного Refresh-токена закінчується (Максимальний вік сесії)';
COMMENT ON COLUMN REFRESH_TOKENS.CREATED_AT IS 'Дата та час ініціалізації сесії (Первинний вхід в систему)';
COMMENT ON COLUMN REFRESH_TOKENS.UPDATED_AT IS 'Дата та час останньої активності або успішного оновлення сесії через тригер';
COMMENT ON COLUMN REFRESH_TOKENS.REVOKED_AT IS 'Дата та час відкликання/заміни сесії. Якщо NULL - сесія вважається активною';
COMMENT ON COLUMN REFRESH_TOKENS.REVOKE_REASON IS 'Причина закриття сесії: logout, rotated (замінено новим), compromised_chain (виявлено повторне використання), expired (протерміновано)';
