-- ============================================================================
-- 2. ТАБЛИЦЯ КАНАЛІВ ВЕРИФІКАЦІЇ (USER_2FA_CHANNELS)
-- ============================================================================
CREATE TABLE USER_2FA_CHANNELS (
    ID             NUMBER GENERATED BY DEFAULT AS IDENTITY,
    USER_ID        NUMBER NOT NULL,
    TYPE           VARCHAR2(20) NOT NULL,
    TARGET         VARCHAR2(100) NOT NULL,
    IS_VERIFIED    CHAR(1) DEFAULT 'N' NOT NULL,
    IS_DEFAULT     CHAR(1) DEFAULT 'N' NOT NULL,
    CREATED_AT     TIMESTAMP DEFAULT CURRENT_TIMESTAMP NOT NULL,
    UPDATED_AT     TIMESTAMP DEFAULT CURRENT_TIMESTAMP NOT NULL,

    CONSTRAINT PK_USER_2FA_CHANNELS PRIMARY KEY (ID),
    CONSTRAINT FK_U2FA_USER FOREIGN KEY (USER_ID) REFERENCES USERS(ID) ON DELETE CASCADE,
    CONSTRAINT CHK_U2FA_VER CHECK (IS_VERIFIED IN ('Y', 'N')),
    CONSTRAINT CHK_U2FA_DEF CHECK (IS_DEFAULT IN ('Y', 'N')),
    CONSTRAINT UK_U2FA_USER_TYPE UNIQUE (USER_ID, TYPE)
);

CREATE INDEX IDX_U2FA_USER ON USER_2FA_CHANNELS(USER_ID);

COMMENT ON TABLE USER_2FA_CHANNELS IS 'Додаткові канали відправки або перевірки одноразових кодів (2FA) для користувачів.';

COMMENT ON COLUMN USER_2FA_CHANNELS.ID IS 'Первинний ключ запису каналу верифікації.';
COMMENT ON COLUMN USER_2FA_CHANNELS.USER_ID IS 'Зовнішній ключ, що вказує на володільця каналу в таблиці USERS.';
COMMENT ON COLUMN USER_2FA_CHANNELS.TYPE IS 'Тип каналу зв''язку для двофакторної перевірки (напр., SMS, EMAIL, TELEGRAM, PUSH).';
COMMENT ON COLUMN USER_2FA_CHANNELS.TARGET IS 'Адресат доставки коду (номер телефону, email, chat_id або токен пристрою).';
COMMENT ON COLUMN USER_2FA_CHANNELS.IS_VERIFIED IS 'Прапорець успішного підтвердження володіння каналом з боку користувача (Y/N).';
COMMENT ON COLUMN USER_2FA_CHANNELS.IS_DEFAULT IS 'Прапорець основного каналу. Тільки один канал певного типу для юзера може бути типовим (Y/N).';
COMMENT ON COLUMN USER_2FA_CHANNELS.CREATED_AT IS 'Дата підключення каналу верифікації.';
COMMENT ON COLUMN USER_2FA_CHANNELS.UPDATED_AT IS 'Дата останнього оновлення параметрів каналу.';


--
CREATE OR REPLACE TRIGGER TRG_U2FA_UPDATED_AT
BEFORE UPDATE ON USER_2FA_CHANNELS
FOR EACH ROW
BEGIN
    :NEW.UPDATED_AT := CURRENT_TIMESTAMP;
END;
/
