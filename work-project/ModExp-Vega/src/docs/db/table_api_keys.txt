-- ============================================================================
-- 4. ТАБЛИЦЯ API КЛЮЧІВ ДЛЯ СТОРІННІХ ІНТЕГРАЦІЙ (API_KEYS)
-- ============================================================================
CREATE TABLE API_KEYS (
    ID                 NUMBER GENERATED BY DEFAULT AS IDENTITY,
    USER_ID            NUMBER NOT NULL,
    CLIENT_NAME        VARCHAR2(100) NOT NULL,
    PREFIX             VARCHAR2(8) NOT NULL,
    HASH               VARCHAR2(64) NOT NULL,
    ALLOWED_IPS        VARCHAR2(1000) DEFAULT NULL,
    IS_ACTIVE          CHAR(1) DEFAULT 'Y' NOT NULL,
    EXPIRES_AT         TIMESTAMP DEFAULT NULL,
    LAST_USED_AT       TIMESTAMP DEFAULT NULL,
    CREATED_AT         TIMESTAMP DEFAULT CURRENT_TIMESTAMP NOT NULL,

    CONSTRAINT PK_API_KEYS PRIMARY KEY (ID),
    CONSTRAINT FK_AK_USER FOREIGN KEY (USER_ID) REFERENCES USERS(ID) ON DELETE CASCADE,
    CONSTRAINT UK_AK_HASH UNIQUE (HASH),
    CONSTRAINT CHK_AK_ACTIVE CHECK (IS_ACTIVE IN ('Y', 'N')),
    CONSTRAINT CHK_AK_IPS CHECK (ALLOWED_IPS IS NULL OR ALLOWED_IPS IS JSON)
);

-- Обов''язковий індекс для миттєвої фільтрації ключів за їхнім публічним початком (префіксом)
CREATE INDEX IDX_AK_PREFIX ON API_KEYS(PREFIX);
CREATE INDEX IDX_AK_USER ON API_KEYS(USER_ID);

COMMENT ON TABLE API_KEYS IS 'Реєстр токенів та ключів доступу для міжсистемної інтеграції (Machine-to-Machine) без участі людини.';

COMMENT ON COLUMN API_KEYS.USER_ID IS 'Посилання на користувача (часто сервісного/технічного акаунта), під чиїм набором ролей виконується інтеграція.';
COMMENT ON COLUMN API_KEYS.CLIENT_NAME IS 'Текстове ім''я або назва сторонньої системи-партнера, яка використовує цей ключ.';
COMMENT ON COLUMN API_KEYS.PREFIX IS 'Відкрита частина ключа (перші 8 символів) для швидкої індексації та візуальної ідентифікації клієнтом.';
COMMENT ON COLUMN API_KEYS.HASH IS 'Криптографічний SHA-256 хеш повної строки секретного ключа. Використовується для фінальної валідації запиту.';
COMMENT ON COLUMN API_KEYS.ALLOWED_IPS IS 'Контекстне обмеження (ABAC): JSON-масив дозволених IP-адрес чи підмереж, з яких сервіс має право слати запити.';
COMMENT ON COLUMN API_KEYS.IS_ACTIVE IS 'Поточний статус працездатності інтеграційного ключа (Y/N).';
COMMENT ON COLUMN API_KEYS.EXPIRES_AT IS 'Гранична дата працездатності ключа. Якщо NULL — ключ вважається безстроковим.';
COMMENT ON COLUMN API_KEYS.LAST_USED_AT IS 'Мітка часу, коли сторонній сервіс востаннє успішно здійснив запит, використовуючи даний ключ.';
